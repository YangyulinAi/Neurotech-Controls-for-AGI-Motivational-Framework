最快落地方案（含最小代码改动草案）
A. 统一“任意格式”的离线分析脚本

新建：tests/analyze_file_onnx.py（替代 analyze_set_file_onnx.py）。

核心思路

探测扩展名：.set 用 mne.io.read_raw_eeglab(preload=True)；.fif 用 mne.io.read_raw_fif(preload=True)；.csv 用 pandas.read_csv（约定列名或首列为时间）。

统一得到 raw.get_data()（形状 channels × time）。

其余流程复用你现有的预处理、特征、ONNX 推理与 /api/bci/broadcast 广播。

Python 关键片段（示意）

# tests/analyze_file_onnx.py
import os, sys, time, json, numpy as np
import requests
import mne
import pandas as pd

sys.path.insert(0, os.path.join(os.path.dirname(__file__), '..', 'src'))
from preprocess import Preprocessor, extract_feats
from onnx_runner import ONNXRunner
from phi_estimator import PhiEstimator

def load_any(filepath, fs_hint=None):
    ext = os.path.splitext(filepath)[1].lower()
    if ext == '.set':
        raw = mne.io.read_raw_eeglab(filepath, preload=True)
    elif ext == '.fif':
        raw = mne.io.read_raw_fif(filepath, preload=True)
    elif ext == '.csv':
        df = pd.read_csv(filepath)
        # 假设列名为通道名或第一列是时间戳
        if 'time' in df.columns[0].lower():
            data = df.iloc[:, 1:].to_numpy().T
        else:
            data = df.to_numpy().T
        sfreq = fs_hint or 256
        info = mne.create_info(ch_names=[f"ch{i}" for i in range(data.shape[0])],
                               sfreq=sfreq, ch_types='eeg')
        raw = mne.io.RawArray(data, info)
    else:
        raise ValueError(f"Unsupported file type: {ext}")
    return raw

def main():
    filepath = sys.argv[1]
    compute_phi = '--compute_phi' in sys.argv
    method = 'mock'
    if '--phi_method' in sys.argv:
        method = sys.argv[sys.argv.index('--phi_method')+1]

    raw = load_any(filepath)
    fs = int(raw.info['sfreq'])
    data = raw.get_data()  # (channels, time)

    pre = Preprocessor(fs, 1, 45)
    onnx = ONNXRunner('model/va_regressor.onnx', use_gpu=True)
    phi_est = PhiEstimator(method=method)

    win_s, step_s = 1.0, 0.25
    win = int(fs * win_s); step = int(fs * step_s)

    for start in range(0, data.shape[1]-win, step):
        seg = data[:, start:start+win]
        seg_t = seg.T
        pre_t = pre.transform(seg_t).T
        spec3, de = extract_feats(pre_t.T, fs)  # 注意：与你项目的函数签名保持一致
        out = onnx.predict(spec3[np.newaxis, ...], de[np.newaxis, :])[0]
        ts = start / fs
        payload = {'valence': float(out[0]), 'arousal': float(out[1]), 'timestamp': float(ts)}
        if compute_phi:
            payload['phi'] = float(phi_est.estimate_phi(pre_t[:8]))  # 取前8通道
        try:
            requests.post('http://localhost:5000/api/bci/broadcast', json=payload, timeout=1)
        except Exception:
            pass
        print(json.dumps(payload))
    print(json.dumps({'type': 'analysis_complete'}))

if __name__ == '__main__':
    main()

const allowedExt = new Set(['.set', '.fif', '.csv']);

const uploadData = multer({
