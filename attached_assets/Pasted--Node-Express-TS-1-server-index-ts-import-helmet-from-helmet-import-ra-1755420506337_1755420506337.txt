1) 统一请求校验 + 限流 + 安全头
// server/index.ts
import helmet from "helmet";
import rateLimit from "express-rate-limit";
import { z } from "zod";

app.use(helmet());
app.use(express.json({ limit: "1mb" })); // 防止 /api/bci/broadcast 过大包
app.use(rateLimit({ windowMs: 60_000, max: 120 })); // 60s 120次

// 统一校验：/api/start-analysis
const startSchema = z.object({
  filename: z.string().min(1),
  computePhi: z.boolean().optional().default(false),
  phiMethod: z.enum(["mock","IIT3.0","IIT4.0_light"]).optional().default("mock"),
  mode: z.enum(["offline","live"]).optional().default("offline")
});
app.post("/api/start-analysis", (req,res,next)=>{
  const p = startSchema.safeParse(req.body);
  if(!p.success) return res.status(400).json({error:"Invalid body", issues:p.error.issues});
  (req as any).validated = p.data;
  next();
});

2) 上传更安全：白名单、大小、路径穿越防护
// server/routes.ts
const allowedExt = new Set([".set",".fif",".csv"]); // 若已实现 .edf/.npz 就加上
const uploadData = multer({
  storage: dataStorage,
  limits: { fileSize: 200 * 1024 * 1024, files: 10 },
  fileFilter: (_req, file, cb) => {
    const ext = path.extname(file.originalname).toLowerCase();
    if (!allowedExt.has(ext)) return cb(new Error("Unsupported file type"));
    if (file.originalname.includes("..") || file.originalname.includes("/")) return cb(new Error("Bad filename"));
    cb(null, true);
  }
});

// 读取文件前做路径穿越防护
function safeJoin(base: string, target: string) {
  const p = path.join(base, target);
  if (!p.startsWith(path.resolve(base))) throw new Error("Path traversal");
  return p;
}

3) 分析任务锁 + 子进程超时与清理

防止用户同时点两次“开始分析”。

// server/analysis.ts
import { spawn } from "child_process";
let running = false;

export function startAnalysis({ filename, computePhi, phiMethod }: any, res: any) {
  if (running) return res.status(409).json({ error: "Analysis already running" });
  running = true;

  const args = ["tests/analyze_file_onnx.py", filename];
  if (computePhi) args.push("--compute_phi","--phi_method", phiMethod);

  const py = spawn(process.env.PYTHON || "python", args, { env: process.env });
  const timer = setTimeout(()=>{ py.kill("SIGKILL"); }, 1000*60*10); // 10 min 兜底

  py.stdout.on("data", (d)=> console.log("[PY]", d.toString().trim()));
  py.stderr.on("data", (d)=> console.error("[PY-ERR]", d.toString().trim()));
  py.on("close", (code)=>{
    clearTimeout(timer);
    running = false;
    console.log("analysis exited with", code);
  });

  return res.json({ ok: true });
}

4) WebSocket 心跳，避免“假在线”
// server/ws.ts
const INTERVAL = 30_000;
wss.on("connection", (ws) => {
  (ws as any).isAlive = true;
  ws.on("pong", ()=> (ws as any).isAlive = true);
});
setInterval(()=>{
  wss.clients.forEach((ws:any)=>{
    if (!ws.isAlive) return ws.terminate();
    ws.isAlive = false; ws.ping();
  });
}, INTERVAL);
